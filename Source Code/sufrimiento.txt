def calcular_doble(numero):
    return numero * 2

def mostrar_mensaje(texto):
    print("Mensaje:", texto)

def main():
    contador = 1

    while contador <= 5:
        print("Iteración:", contador)

        # Uso de if
        if contador % 2 == 0:
            mostrar_mensaje("El número es par")
        else:
            mostrar_mensaje("El número es impar")

        resultado = calcular_doble(contador)
        print("El doble es:", resultado)

        contador += 1

main()
