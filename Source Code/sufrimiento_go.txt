package main

import "fmt"
func calcularDoble(numero int) int {
	return numero * 2
}

func mostrarMensaje(texto string) {
	fmt.Println("Mensaje:", texto)
}

func main() {
	contador := 1
	for contador <= 5 {
		fmt.Println("Iteración:", contador)

		if contador%2 == 0 {
			mostrarMensaje("El número es par")
		} else {
			mostrarMensaje("El número es impar")
		}

		resultado := calcularDoble(contador)
		fmt.Println("El doble es:", resultado)
		contador++
	}
}
